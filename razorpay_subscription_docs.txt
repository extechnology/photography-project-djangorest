5. RAZORPAY WEBHOOK (ASYNCHRONOUS SERVER-TO-SERVER)
--------------------------------------------------------------------------------
URL:
  POST /api/plans/webhook/
  POST /api/subscriptions/webhook/

Auth:
  None (Public endpoint, verified cryptographically via header)

Headers Required:
  X-Razorpay-Signature: <hmac_sha256_hex_digest>
  Content-Type: application/json

Events Handled:
  1. "payment.captured" / "order.paid":
     - Finds pending SubscriptionPayment by order_id
     - Marks payment status as 'success'
     - Activates PhotographerSubscription and applies storage quota immediately
  2. "payment.failed":
     - Updates SubscriptionPayment status to 'failed'

Response (200 OK):
{
  "status": "success",
  "message": "Subscription activated for order order_NXK54xL09XjL9K"
}

8. FRONTEND CHECKOUT JS CODE SNIPPET (REACT / VANILLA JS)
--------------------------------------------------------------------------------
async function handleSubscribe(planId) {
  // Step 1: Request Order ID from backend
  const checkoutRes = await fetch('/api/plans/checkout/', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${accessToken}`
    },
    body: JSON.stringify({ plan_id: planId, gateway: 'razorpay' })
  });
  const orderData = await checkoutRes.json();

  // Step 2: Open Razorpay Checkout Modal
  const options = {
    key: orderData.key_id,
    amount: orderData.amount_paise,
    currency: orderData.currency,
    name: 'EX SHARE',
    description: 'Studio Tier Subscription',
    order_id: orderData.order_id,
    handler: async function (response) {
      // Step 3: Verify signature on backend
      const verifyRes = await fetch('/api/plans/verify/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${accessToken}`
        },
        body: JSON.stringify({
          plan_id: planId,
          razorpay_order_id: response.razorpay_order_id,
          razorpay_payment_id: response.razorpay_payment_id,
          razorpay_signature: response.razorpay_signature
        })
      });
      const verifyData = await verifyRes.json();
      if (verifyData.status === 'success') {
        alert('Subscription activated successfully!');
        window.location.reload();
      } else {
        alert('Payment verification failed.');
      }
    },
    theme: {
      color: '#000000'
    }
  };

  const rzp = new window.Razorpay(options);
  rzp.open();
}
